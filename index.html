const crypto = require('crypto'); const fs = require('fs'); const pdfData = fs.readFileSync('resume.pdf'); const pdfBase64 = pdfData.toString('base64'); const PASSWORD = '1234567890'; // same password as in your HTML const SALT = crypto.randomBytes(16); const IV = crypto.randomBytes(12); // ✅ Use PBKDF2 here — matches what the browser uses crypto.pbkdf2(PASSWORD, SALT, 100000, 32, 'sha256', (err, KEY) => { if (err) throw err; const cipher = crypto.createCipheriv('aes-256-gcm', KEY, IV); const enc = Buffer.concat([cipher.update(pdfBase64, 'utf8'), cipher.final()]); const authTag = cipher.getAuthTag(); // Pack: salt(16) + iv(12) + authTag(16) + encryptedData const packed = Buffer.concat([SALT, IV, authTag, enc]).toString('base64'); console.log('\n=== PASTE THIS AS RESUME_ENC in your HTML ===\n'); console.log(packed); console.log('\n=============================================\n'); });
